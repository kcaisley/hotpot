\section{Export}

\begin{frame}[fragile]{Export the final design}
\slidesource{\href{https://github.com/The-OpenROAD-Project/OpenROAD/blob/master/src/README.md}{OpenROAD write commands}}
\begin{lstlisting}
# OpenROAD database (.odb)
write_db final.odb
# Routed layout and placement (.def)
write_def final.def
# Cell and pin abstract (.lef)
write_abstract_lef final.lef
# Gate-level netlist (.v)
write_verilog -remove_cells {FILLCELL*} final.v
# Circuit netlist for LVS (.cdl); cells.cdl is the PDK master
write_cdl -masters {cells.cdl} final.cdl
# Timing constraints (.sdc)
write_sdc final.sdc
\end{lstlisting}
\end{frame}

\vspace{4pt}
\begin{lstlisting}
# Replace generic filenames and top with your design inputs.
klayout -b -r scripts/def2stream.py \
  -rd tech_file=platform.lyt \
  -rd layer_map= \
  -rd in_def=final.def \
  -rd design_name=top \
  -rd in_files="cells.gds macros.gds" \
  -rd seal_file= \
  -rd out_file=final.gds
\end{lstlisting}
\end{frame}
