\begin{column}{.50\textwidth}
\begin{lstlisting}
# Check VDD/VSS connectivity. check_power_grid alone does not
# populate IR Drop. The optional solver writes voltage/error data.
check_power_grid -net VDD
check_power_grid -net VSS
analyze_power_grid -net VDD \\
  -source_type FULL \\
  -voltage_file results/ariane/vdd.voltage.rpt \\
  -error_file results/ariane/vdd.errors.rpt
\end{lstlisting}
