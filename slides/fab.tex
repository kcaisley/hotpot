\newcommand{\waferdrawing}[1]{%
\begin{tikzpicture}[scale=1.02]
  \fill[NordPanel] (0,0) circle (2.33);
  \begin{scope}
    \clip (0,0) circle (2.33);
    \foreach \ix in {0,...,5}{
      \foreach \iy in {0,...,5}{
        \pgfmathsetmacro{\xx}{-2.4+0.8*\ix}
        \pgfmathsetmacro{\yy}{-2.4+0.8*\iy}
        \ifnum#1=1
          \fill[NordPurple] (\xx,\yy) rectangle ++(.77,.77);
        \else
          \fill[NordBlue] (\xx,\yy) rectangle ++(.77,.77);
        \fi
        \draw[NordBody!55,line width=.45pt] (\xx,\yy) rectangle ++(.77,.77);
        \ifnum#1=1
          \draw[NordBody!90,dotted,line width=.65pt] (\xx+.04,\yy+.04) rectangle ++(.69,.69);
        \fi
        \ifnum#1=2
          \draw[NordBody!90,dotted,line width=.65pt] (\xx+.385,\yy) -- ++(0,.77);
          \draw[NordBody!90,dotted,line width=.65pt] (\xx,\yy+.385) -- ++(.77,0);
          \fill[NordPurple] (\xx+.04,\yy+.43) rectangle ++(.30,.30);
        \fi
        \ifnum#1=3
          \draw[NordBody!90,dotted,line width=.65pt] (\xx+.055,\yy+.055) rectangle ++(.66,.66);
          \fill[NordPurple] (\xx+.10,\yy+.48) rectangle ++(.12,.12);
        \fi
      }
    }
  \end{scope}
  \draw[NordBlue,line width=1.1pt] (0,0) circle (2.33);
\end{tikzpicture}%
}
