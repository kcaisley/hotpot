\begin{frame}[fragile]{Define the timing contract}
\vfill
\begin{lstlisting}
# Define a 0.510 ns core clock.
create_clock -name core_clock -period 0.510 \
  [get_ports clk]

# Reserve 0.102 ns at each input.
# Exclude the clock port itself.
set_input_delay 0.102 -clock core_clock \
  [lsearch -inline -all -not -exact \
   [all_inputs] [get_ports clk]]

# Reserve 0.102 ns at each output.
set_output_delay 0.102 -clock core_clock \
  [all_outputs]
\end{lstlisting}
\vfill
\end{frame}

\section{Clock-tree synthesis}

\begin{frame}[fragile]{Clock-tree synthesis and timing repair}
\begin{columns}[T,onlytextwidth]
\begin{column}{.47\textwidth}\centering
\includegraphics[width=\linewidth,height=.80\textheight,keepaspectratio]{images/routing/gcd_clock_tree.png}
\end{column}
\begin{column}{.50\textwidth}
\begin{lstlisting}
# Normalize pre-existing clock inverters so CTS sees a consistent tree.
repair_clock_inverters
clock_tree_synthesis -root_buf BUF_X4 \\
  -buf_list BUF_X4 -sink_clustering_enable \\
  -sink_clustering_max_diameter 100
repair_clock_nets
# Legalize CTS cells; timing repair may move or insert cells, so
# detailed placement is repeated after repair_timing.
detailed_placement
set_propagated_clock [all_clocks]
estimate_parasitics -placement
repair_timing -skip_gate_cloning
detailed_placement
\end{lstlisting}
\end{column}
\end{columns}
\end{frame}

\begin{lstlisting}
# Draw the worst setup path in the layout. Clicking a Timing Report
# row selects that row's path; report_checks prints the path in text.
# Launch: source register. Capture: destination register.
# Signal: data path. Inst: cell. Term: pin or terminal.
gui::show_worst_path -setup
report_checks -path_delay max \\
  -format full_clock_expanded \\
  -fields {input_pin slew capacitance} -digits 3
\end{lstlisting}
