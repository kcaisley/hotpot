\begin{frame}{Four ways to do global placement}
\centering

\textcolor{NordBlue}{\textbf{Mode}} &
\textcolor{NordBlue}{\textbf{Pins}} &
\textcolor{NordBlue}{\textbf{Needs}} \\
\hline
\texttt{global\_placement} &
Uses placed pins; \texttt{-skip\_io} ignores them &
Floorplan, cells, and rows \\
\texttt{-place\_ios} &
Co-optimizes movable pins and cells &
Supported die and pin setup; run \texttt{place\_pins} afterward \\
\texttt{-routability\_driven} &
Uses placed pins &
Routing layers for congestion estimates; no prior routing or parasitic estimate \\
\texttt{-timing\_driven} &
Uses placed pins &
Liberty, timing constraints, and wire RC; CTS is not required
\end{tabular}

# One set_layer_rc call per layer.
# -layer takes one name, not a range.
# Tcl foreach can reuse R/C values
# across groups of layers.
# E.g. M1-M4 and M6-M7, separately.
# Override missing or unsuitable
# LEF values only.
set_layer_rc -layer metal3 \
  -resistance 5e-3 \
  -capacitance 5e-2
# set_wire_rc chooses signal and
# clock layers for wire estimates.
set_wire_rc -signal -layer metal3
set_wire_rc -clock -layer metal6
# Use placed pin locations, not
# detailed routed wires.
estimate_parasitics -placement
\end{lstlisting}

\section{Legalization}

\begin{frame}{Placement engines}
\begin{itemize}
\setlength{\itemsep}{.65em}
\item Supports mixed cell heights (1x--4x)
\item Supports fence regions and fragmented rows
\item Supports placement padding and filler insertion
\item Two legalization engines:
  \begin{itemize}
  \setlength{\itemsep}{.35em}
  \item \textbf{Negotiation legalizer (default):} handles dense placement,
    but is slower; uses diamond search for remaining corner cases.
  \item \textbf{Diamond search (legacy):} faster search from each cell's
    starting position. Enable with \texttt{-use\_diamond\_legalizer}.
  \end{itemize}
\end{itemize}
\end{frame}
