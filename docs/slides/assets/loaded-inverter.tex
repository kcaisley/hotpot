\documentclass[border=10pt]{standalone}
\usepackage[american]{circuitikz}
\usetikzlibrary{calc}
\usepackage{lmodern}
\definecolor{NordBody}{HTML}{4C566A}
\definecolor{NordBlue}{HTML}{5E81AC}
\definecolor{NordPurple}{HTML}{B48EAD}
\begin{document}
\begin{circuitikz}[color=NordBody,line width=.8pt,font=\sffamily]
\node[not port,scale=1.6] (inv) at (5.4,2) {};
\draw (0,2) to[V] (0,0) node[ground]{};
\node[anchor=east] at (-.5,1) {$V_{src}$};
\draw (0,2) to[R,l={$R_S=2\,\mathrm{k}\Omega$}] (2.8,2) coordinate (A) -- (inv.in);
\draw (A) to[C,l_={$C_{in}=5\,\mathrm{fF}$}] (2.8,0) node[ground]{};
\node[above,text=NordBlue] at (A) {A};
\draw (inv.out) -- (7,2) coordinate (ZN)
 to[R,l={$R_L=1\,\mathrm{k}\Omega$}] (10,2) coordinate (load)
 to[C,l={$C_L=10\,\mathrm{fF}$}] (10,0) node[ground]{};
\node[above,text=NordBlue] at (ZN) {ZN};
\node[above,text=NordPurple] at (load) {load};
% Explicit supply pins on the two sloping edges of the inverter triangle.
\coordinate (tip) at ($(inv.center)!.55!(inv.east)$);
\coordinate (upper) at ($(inv.bin |- inv.north)!.5!(tip)$);
\coordinate (lower) at ($(inv.bin |- inv.south)!.5!(tip)$);
\coordinate (vdd) at (upper |- 0,4);
\draw[color=black] (upper) -- (vdd);
\draw[color=black,line width=1.4pt] ($(vdd)+(-.38,0)$) -- ($(vdd)+(.38,0)$);
\node[above=4pt,text=black] at (vdd) {$V_{DD}=1.10\,\mathrm{V}$};
\draw[-latex,color=NordBlue] ($(upper)+(.5,1.3)$) -- ($(upper)+(.5,.65)$) node[midway,right]{$I_{DD}$};
\draw (lower) -- (lower |- 0,.2) node[ground]{} node[right=5pt] {VSS};
\node[anchor=south east,text=black] at ($(inv.bin |- inv.north)+(-.1,.12)$) {INV\_X1};
\node[text=NordBlue] at (5.4,-1.3) {Cell delay: A $\rightarrow$ ZN \qquad Load delay: ZN $\rightarrow$ load};
\end{circuitikz}
\end{document}
